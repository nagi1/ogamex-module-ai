    /**
     * The foreign planets worth a probe, chosen by relevance in SQL before the cap
     * (ATK-001): the owners least recently seen first — the host's own last-login
     * stamp, so an account that stopped logging in leads and one seen this minute
     * trails — and, among those, the ones nearest the idle probe's origin. Taking
     * the lowest ids instead takes the oldest planets in the universe, the accounts
     * already looked at, so a neighbour the universe creates later is never a
     * candidate and no probe is ever planned where it matters.
     *
     * @return Collection<int, Planet>
     */
    private function relevantCandidates(PlayerService $player, Coordinate $anchor): Collection
    {
        return Planet::query()
            ->leftJoin('users', 'users.id', '=', 'planets.user_id')
            ->where('planets.user_id', '!=', $player->getId())
            ->where('planets.destroyed', 0)
            // A zero stamp records no visit at all, so it is no evidence of an idle
            // owner: it ranks behind every planet whose owner has a real last-login.
            ->orderByRaw('case when users.time is null or users.time = 0 then 1 else 0 end')
            ->orderBy('users.time')
            ->orderByRaw('case when planets.galaxy = ? then 0 else 1 end', [$anchor->galaxy])
            ->orderByRaw('abs(planets.system - ?)', [$anchor->system])
            ->orderByRaw('abs(planets.planet - ?)', [$anchor->position])
            ->orderBy('planets.id')
            ->limit(self::MAX_CANDIDATES)
            ->get(['planets.*']);
    }
