### FILE: resources/behavior/social_contact.yaml
```yaml
# Contact an account opens on its own initiative, rather than in answer to an arrival.
# Every answered exchange is one cue that may reach out to the counterparty; these two
# numbers decide whether it does.
#
# open_contact_limit: how many exchanges between the two may still wait for an answer
# before the account stops opening new ones. The cap wins over freshness.
open_contact_limit: 1
#
# cue_max_age_hours: how long an answered exchange stays a live conversation. An answer
# given long after the arrival is history, not contact, and never opens anything.
cue_max_age_hours: 24
```

### FILE: tests/Feature/SocialContactInitiationTest.php
```php
<?php

use Carbon\CarbonImmutable;
use Modules\AI\Actions\EvaluateAiSocialExchangeAction;
use Modules\AI\Contracts\SocialCognition;
use Modules\AI\Domain\Conversation\NativeSocialCognition;
use Modules\AI\Enums\AiSocialExchangeState;
use Modules\AI\Enums\AiSocialExchangeType;
use Modules\AI\Models\AiSocialExchange;
use Modules\AI\Tests\Support\AiQueueModuleTestCase;

uses(AiQueueModuleTestCase::class);

const SOCIAL_CONTACT_NOW = '2026-09-30 12:00:00 UTC';

beforeEach(function (): void {
    app()->bind(SocialCognition::class, NativeSocialCognition::class);
});

function socialContactCue(AiSocialExchangeType $type, CarbonImmutable $createdAt, int $sourceObservationId = 7): AiSocialExchange
{
    return AiSocialExchange::create([
        'player_id' => 1,
        'counterparty_player_id' => 2,
        'source_observation_id' => $sourceObservationId,
        'type' => $type,
        'terms' => [],
        'state' => AiSocialExchangeState::Proposed,
        'revision' => 1,
        'created_at' => $createdAt,
    ]);
}

function socialContactOpenings(int $cueId): int
{
    return AiSocialExchange::query()
        ->where('type', AiSocialExchangeType::Greeting)
        ->whereKeyNot($cueId)
        ->count();
}

test('an account that is only ever answering opens contact of its own', function (): void {
    $now = CarbonImmutable::parse(SOCIAL_CONTACT_NOW);
    $cue = socialContactCue(AiSocialExchangeType::Thanks, $now);

    $answered = app(EvaluateAiSocialExchangeAction::class)->handle($cue->id, 0.0, $now);

    expect($answered->state)->toBe(AiSocialExchangeState::Responded)
        ->and($answered->responded_at)->not->toBeNull();

    $opening = AiSocialExchange::query()->where('type', AiSocialExchangeType::Greeting)->first();

    expect($opening)->not->toBeNull()
        ->and($opening->player_id)->toBe(1)
        ->and($opening->counterparty_player_id)->toBe(2)
        ->and($opening->state)->toBe(AiSocialExchangeState::Proposed)
        ->and($opening->revision)->toBe(1);
});

test('a cue answered right at the end of its freshness window still opens contact', function (): void {
    $now = CarbonImmutable::parse(SOCIAL_CONTACT_NOW);
    $cue = socialContactCue(AiSocialExchangeType::Thanks, $now->subHours(24));

    app(EvaluateAiSocialExchangeAction::class)->handle($cue->id, 0.0, $now);

    expect(socialContactOpenings($cue->id))->toBe(1);
});

test('an exchange answered past its freshness window is history, not an opening', function (): void {
    $now = CarbonImmutable::parse(SOCIAL_CONTACT_NOW);
    $cue = socialContactCue(AiSocialExchangeType::Thanks, $now->subHours(25));

    $answered = app(EvaluateAiSocialExchangeAction::class)->handle($cue->id, 0.0, $now);

    expect($answered->state)->toBe(AiSocialExchangeState::Responded)
        ->and(socialContactOpenings($cue->id))->toBe(0);
});

test('an answer still owed between the two stops a reach-out, at the cap and past it', function (int $pending): void {
    $now = CarbonImmutable::parse(SOCIAL_CONTACT_NOW);

    foreach (range(1, $pending) as $index) {
        socialContactCue(AiSocialExchangeType::Thanks, $now->subHours(2), 7 + $index);
    }

    $cue = socialContactCue(AiSocialExchangeType::Thanks, $now, 7);

    $answered = app(EvaluateAiSocialExchangeAction::class)->handle($cue->id, 0.0, $now);

    expect($answered->state)->toBe(AiSocialExchangeState::Responded)
        ->and(socialContactOpenings($cue->id))->toBe(0);
})->with([
    'at the cap' => 1,
    'past the cap' => 2,
]);
```
