    /**
     * A login's one errand. ScheduleAiIntentAction refills the build queues and the lab in every
     * session whatever wins the slot, so a chore that fills them anyway spends the errand on work
     * that was going to happen — and the fleet errand the situation actually offers (a raid from a
     * report, a look at a neighbour, debris at home, a colony ship, an expedition) never gets picked.
     * The fleet comes first even when the shipyard has something to order: an account holding a
     * report or a neighbour worth watching spends its login on the fleet, not on another cargo batch.
     * Only when the situation offers no fleet errand does the best remaining candidate take the slot.
     * Either way doing nothing is a floor: a quiet login is still a legitimate choice.
     *
     * @param array<int, ScoredCandidate> $scored
     */
    private function situationalErrand(array $scored, ScoredCandidate $selected): ScoredCandidate
    {
        if (!$this->isEconomyChore($scored[0]->candidate->type)) {
            return $selected;
        }

        $doNothing = $this->doNothingAmong($scored);
        if ($doNothing === null) {
            return $selected;
        }

        // $scored is already ranked best first, so the first fleet errand is the best one offered.
        $fallback = null;
        foreach ($scored as $candidate) {
            $type = $candidate->candidate->type;
            if ($this->isEconomyChore($type)) {
                continue;
            }

            $fallback ??= $candidate;
            if ($this->isFleetErrand($type)) {
                return $candidate->score > $doNothing->score ? $candidate : $selected;
            }
        }

        return $fallback !== null && $fallback->score > $doNothing->score ? $fallback : $selected;
    }
