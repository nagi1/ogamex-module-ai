    private function commitmentDirection(AiSocialExchange $exchange): AiCommitmentDirection
    {
        return match ($exchange->type) {
            AiSocialExchangeType::HelpRequest => AiCommitmentDirection::PromisedByPlayer,
            AiSocialExchangeType::CompensationOffer => AiCommitmentDirection::ExpectedFromCounterparty,
            default => throw app()->makeWith(LogicException::class, ['message' => 'Only commitment-producing exchanges have a direction.']),
        };
    }

    /**
     * An answered exchange that nobody ever comes back to is how a lane reads as health while
     * nothing happens. Once the pair has been quiet for the whole window the account takes the
     * contact back into its own hands: the exchange is re-opened as a fresh proposal, so a cohort
     * run shows new exchange activity and a relationship that moves with it.
     *
     * An exchange that already produced a promise is left alone — re-opening it would erase what
     * was agreed, and the commitment lane owns what happens next.
     */
    private function reengageQuietContact(AiSocialExchange $exchange, CarbonImmutable $evaluatedAt): AiSocialExchange
    {
        if ($exchange->commitment_id !== null) {
            return $exchange;
        }

        if (!$this->readsAsDormant($exchange, $evaluatedAt)) {
            return $exchange;
        }

        $exchange->update([
            'state' => AiSocialExchangeState::Proposed,
            'response' => null,
            'response_terms' => null,
            'response_reason' => null,
            'responded_at' => null,
            'due_at' => $evaluatedAt->addHours($this->dormancyWindowHours()),
            'revision' => $exchange->revision + 1,
        ]);

        $this->refreshPairContact($exchange);

        return $exchange->refresh();
    }

    /**
     * Silence stays healthy only while it is short. The window is measured from the last contact —
     * the answer the pair actually exchanged — so an old exchange answered a moment ago is live.
     */
    private function readsAsDormant(AiSocialExchange $exchange, CarbonImmutable $evaluatedAt): bool
    {
        $lastContact = $exchange->responded_at ?? $exchange->due_at ?? $exchange->created_at;

        if ($lastContact === null) {
            return false;
        }

        return CarbonImmutable::instance($lastContact)
            ->addHours($this->dormancyWindowHours())
            ->lessThanOrEqualTo($evaluatedAt);
    }

    /**
     * The pair's relationship row moves with the contact, so the cohort window shows an account
     * that is socially alive rather than a lane that only ever reports absence of activity.
     */
    private function refreshPairContact(AiSocialExchange $exchange): void
    {
        AiRelationship::query()
            ->where('player_id', $exchange->player_id)
            ->where('other_player_id', $exchange->counterparty_player_id)
            ->first()
            ?->touch();
    }

    /**
     * The behaviour file holds scalar keys, so it is read line-wise instead of pulling a parser
     * into a decision. The key lives in the data file and nowhere else.
     */
    private function dormancyWindowHours(): int
    {
        $lines = file(dirname(__DIR__, 2) . self::SOCIAL_COGNITION_DATA_FILE, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: [];
        $pattern = '/^\s*' . self::DORMANCY_WINDOW_KEY . ':\s*(\d+)\s*$/';

        foreach ($lines as $line) {
            if (preg_match($pattern, $line, $matches) === 1) {
                return (int) $matches[1];
            }
        }

        throw app()->makeWith(LogicException::class, ['message' => 'The social cognition data file must state ' . self::DORMANCY_WINDOW_KEY . '.']);
    }

<?php

use Carbon\CarbonImmutable;
use Modules\AI\Actions\EvaluateAiSocialExchangeAction;
use Modules\AI\Enums\AiSocialExchangeState;
use Modules\AI\Enums\AiSocialExchangeType;
use Modules\AI\Enums\AiSocialResponse;
use Modules\AI\Models\AiSocialExchange;
use Modules\AI\Tests\Support\AiQueueModuleTestCase;

require_once __DIR__ . '/../Support/AiQueueModuleTestCase.php';

uses(AiQueueModuleTestCase::class);

/** The window the social cognition behaviour file states, in hours. */
const SOCIAL_DORMANCY_WINDOW_HOURS = 24;

/**
 * An exchange the account has already answered and left no promise on: the only shape the
 * dormancy read applies to. It carries the observation it was raised from, because an exchange
 * without one is not a record of anything the account saw.
 */
function answeredExchange(array $attributes = []): AiSocialExchange
{
    static $sourceObservationId = 0;
    $sourceObservationId++;

    return AiSocialExchange::create(array_merge([
        'player_id' => 1,
        'counterparty_player_id' => 2,
        'source_observation_id' => $sourceObservationId,
        'type' => AiSocialExchangeType::HelpRequest,
        'terms' => ['metal' => 500],
        'state' => AiSocialExchangeState::Responded,
        'response' => AiSocialResponse::Accept,
        'response_terms' => null,
        'response_reason' => 'nothing owed',
        'due_at' => null,
        'responded_at' => CarbonImmutable::parse('2024-01-01 00:00:00 UTC'),
        'commitment_id' => null,
        'revision' => 2,
    ], $attributes));
}

test('a counterparty quiet past the dormancy window is re-engaged by the account itself', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    $exchange = answeredExchange(['responded_at' => $answeredAt]);
    $evaluatedAt = $answeredAt->addHours(SOCIAL_DORMANCY_WINDOW_HOURS + 6);

    $reengaged = app(EvaluateAiSocialExchangeAction::class)->handle($exchange->id, 0.0, $evaluatedAt);

    expect($reengaged->state)->toBe(AiSocialExchangeState::Proposed)
        ->and($reengaged->responded_at)->toBeNull()
        ->and($reengaged->response)->toBeNull()
        ->and($reengaged->revision)->toEqual(3)
        ->and($reengaged->due_at?->utc()->toDateTimeString())
        ->toBe($evaluatedAt->addHours(SOCIAL_DORMANCY_WINDOW_HOURS)->utc()->toDateTimeString());
});

test('the window is the bound: the moment it has elapsed the contact reads as dormant', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    $exchange = answeredExchange(['responded_at' => $answeredAt]);

    $reengaged = app(EvaluateAiSocialExchangeAction::class)->handle(
        $exchange->id,
        0.0,
        $answeredAt->addHours(SOCIAL_DORMANCY_WINDOW_HOURS),
    );

    expect($reengaged->state)->toBe(AiSocialExchangeState::Proposed)
        ->and($reengaged->revision)->toEqual(3);
});

test('a contact one hour inside the window is left alone', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    $exchange = answeredExchange(['responded_at' => $answeredAt]);

    $untouched = app(EvaluateAiSocialExchangeAction::class)->handle(
        $exchange->id,
        0.0,
        $answeredAt->addHours(SOCIAL_DORMANCY_WINDOW_HOURS - 1),
    );

    expect($untouched->state)->toBe(AiSocialExchangeState::Responded)
        ->and($untouched->revision)->toEqual(2)
        ->and($untouched->responded_at?->utc()->toDateTimeString())->toBe($answeredAt->utc()->toDateTimeString());
});

test('a contact answered this instant is not dormant at zero elapsed time', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    $exchange = answeredExchange(['responded_at' => $answeredAt]);

    $untouched = app(EvaluateAiSocialExchangeAction::class)->handle($exchange->id, 0.0, $answeredAt);

    expect($untouched->state)->toBe(AiSocialExchangeState::Responded)
        ->and($untouched->revision)->toEqual(2)
        ->and($untouched->responded_at)->not->toBeNull();
});

test('an exchange that produced a promise is left to the commitment lane', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    // A promise already owns this exchange, so re-opening the contact would erase what was agreed.
    $exchange = answeredExchange([
        'responded_at' => $answeredAt,
        'commitment_id' => 7,
    ]);

    $untouched = app(EvaluateAiSocialExchangeAction::class)->handle(
        $exchange->id,
        0.0,
        $answeredAt->addHours(SOCIAL_DORMANCY_WINDOW_HOURS * 4),
    );

    expect($untouched->state)->toBe(AiSocialExchangeState::Responded)
        ->and($untouched->revision)->toEqual(2);
});

test('dormancy is read from the answer, not from a due date that has long passed', function (): void {
    $answeredAt = CarbonImmutable::parse('2024-01-01 00:00:00 UTC');
    $exchange = answeredExchange([
        'responded_at' => $answeredAt,
        'due_at' => $answeredAt->subHours(48),
    ]);

    $untouched = app(EvaluateAiSocialExchangeAction::class)->handle(
        $exchange->id,
        0.0,
        $answeredAt->addHours(2),
    );

    expect($untouched->state)->toBe(AiSocialExchangeState::Responded)
        ->and($untouched->revision)->toEqual(2);
});
