    /**
     * An answered message is contact, so the counterparty's freshness moves even when the
     * answer turns out to be a rejection.
     */
    private function refreshRelationship(AiSocialExchange $exchange, CarbonImmutable $evaluatedAt): void
    {
        AiRelationship::query()
            ->where('player_id', $exchange->player_id)
            ->where('other_player_id', $exchange->counterparty_player_id)
            ->update(['updated_at' => $evaluatedAt]);
    }

    /**
     * Answering is the only social work this lane reaches on its own, so an account nobody
     * writes to stays silent for as long as the silence lasts. A counterparty that has been
     * quiet for longer than the contact policy allows is contacted again by the answer.
     */
    private function openContactWhenDormant(AiSocialExchange $exchange, CarbonImmutable $evaluatedAt): void
    {
        $policy = Yaml::parseFile(module_path('AI', 'resources/behavior/social_contact.yaml'));

        if (! $this->hasGoneQuiet($exchange, $evaluatedAt, (int) $policy['dormancy_window_hours'])) {
            return;
        }

        if ($this->contactAlreadyOpen($exchange)) {
            return;
        }

        AiSocialExchange::query()->firstOrCreate(
            [
                'player_id' => $exchange->player_id,
                'source_observation_id' => $exchange->source_observation_id,
                'type' => constant(AiSocialExchangeType::class . '::' . $policy['initiation_type']),
            ],
            [
                'counterparty_player_id' => $exchange->counterparty_player_id,
                'terms' => [],
                'state' => AiSocialExchangeState::Proposed,
                'revision' => 0,
            ],
        );
    }

    private function hasGoneQuiet(AiSocialExchange $exchange, CarbonImmutable $evaluatedAt, int $windowHours): bool
    {
        $lastContact = AiSocialExchange::query()
            ->where('player_id', $exchange->player_id)
            ->where('counterparty_player_id', $exchange->counterparty_player_id)
            ->whereKeyNot($exchange->getKey())
            ->max('responded_at');

        if ($lastContact === null) {
            return true;
        }

        return CarbonImmutable::parse((string) $lastContact)->lessThan($evaluatedAt->subHours($windowHours));
    }

    private function contactAlreadyOpen(AiSocialExchange $exchange): bool
    {
        return AiSocialExchange::query()
            ->where('player_id', $exchange->player_id)
            ->where('counterparty_player_id', $exchange->counterparty_player_id)
            ->where('state', AiSocialExchangeState::Proposed)
            ->exists();
    }

<?php

use Carbon\CarbonImmutable;
use Modules\AI\Actions\EvaluateAiSocialExchangeAction;
use Modules\AI\Contracts\SocialCognition;
use Modules\AI\Domain\Conversation\NativeSocialCognition;
use Modules\AI\Enums\AiSocialExchangeState;
use Modules\AI\Enums\AiSocialExchangeType;
use Modules\AI\Enums\AiSocialResponse;
use Modules\AI\Models\AiSocialExchange;
use Modules\AI\Tests\Support\AiQueueModuleTestCase;

require_once __DIR__ . '/../Support/AiQueueModuleTestCase.php';

uses(AiQueueModuleTestCase::class);

const SOCIAL_DORMANCY_NOW = '2026-09-30 12:00:00 UTC';

beforeEach(function (): void {
    app()->bind(SocialCognition::class, NativeSocialCognition::class);
});

/**
 * An exchange as the module stores it: something a counterparty wrote, answered or not.
 */
function socialExchangeRow(int $observationId, AiSocialExchangeState $state, CarbonImmutable|null $respondedAt = null): AiSocialExchange
{
    return AiSocialExchange::create([
        'player_id' => 1,
        'counterparty_player_id' => 2,
        'source_observation_id' => $observationId,
        'type' => AiSocialExchangeType::Thanks,
        'terms' => [],
        'state' => $state,
        'response' => $state === AiSocialExchangeState::Responded ? AiSocialResponse::Accept : null,
        'responded_at' => $respondedAt,
        'revision' => 1,
    ]);
}

function openedContactCount(): int
{
    return AiSocialExchange::query()
        ->where('counterparty_player_id', 2)
        ->where('type', AiSocialExchangeType::Greeting)
        ->count();
}

test('a counterparty never heard from is contacted by the answer', function (): void {
    $inbound = socialExchangeRow(901, AiSocialExchangeState::Proposed);

    app(EvaluateAiSocialExchangeAction::class)->handle($inbound->id, 0.0, CarbonImmutable::parse(SOCIAL_DORMANCY_NOW));

    $opened = AiSocialExchange::query()->where('type', AiSocialExchangeType::Greeting)->first();

    expect($inbound->refresh()->state)->toBe(AiSocialExchangeState::Responded)
        ->and($opened)->not->toBeNull()
        ->and($opened->state)->toBe(AiSocialExchangeState::Proposed)
        ->and($opened->revision)->toBe(0)
        ->and($opened->source_observation_id)->toBe(901);
});

test('a counterparty answered inside the window is left alone', function (int $hoursAgo): void {
    $now = CarbonImmutable::parse(SOCIAL_DORMANCY_NOW);
    socialExchangeRow(902, AiSocialExchangeState::Responded, $now->subHours($hoursAgo));
    $inbound = socialExchangeRow(903, AiSocialExchangeState::Proposed);

    app(EvaluateAiSocialExchangeAction::class)->handle($inbound->id, 0.0, $now);

    expect(openedContactCount())->toBe(0);
})->with([0, 23, 24]);

test('a counterparty silent past the window is contacted again', function (): void {
    $now = CarbonImmutable::parse(SOCIAL_DORMANCY_NOW);
    socialExchangeRow(904, AiSocialExchangeState::Responded, $now->subHours(25));
    $inbound = socialExchangeRow(905, AiSocialExchangeState::Proposed);

    app(EvaluateAiSocialExchangeAction::class)->handle($inbound->id, 0.0, $now);

    expect(openedContactCount())->toBe(1);
});

test('an already open contact is not opened twice', function (): void {
    $now = CarbonImmutable::parse(SOCIAL_DORMANCY_NOW);
    $first = socialExchangeRow(906, AiSocialExchangeState::Proposed);
    $second = socialExchangeRow(907, AiSocialExchangeState::Proposed);

    app(EvaluateAiSocialExchangeAction::class)->handle($first->id, 0.0, $now);
    app(EvaluateAiSocialExchangeAction::class)->handle($second->id, 0.0, $now);

    expect(openedContactCount())->toBe(1);
});
