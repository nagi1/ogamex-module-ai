<?php

use Modules\AI\Domain\Decision\QueueableRaid;
use Modules\AI\Domain\Decision\RaidPlanner;
use Modules\AI\Models\AiProfile;
use OGame\Models\Enums\PlanetType;
use OGame\Models\EspionageReport;
use OGame\Models\Planet;
use OGame\Models\Resources;
use OGame\Models\User;
use Tests\IsolatedAccountTestCase;

uses(IsolatedAccountTestCase::class);

/**
 * The host's own mid-phase marker: the account owns a second planet. It carries
 * no fleet, so the origin stays the planet the account already flies from.
 */
function activeNeighbourColony(int $playerId): Planet
{
    return Planet::factory()->create([
        'user_id' => $playerId,
        'galaxy' => 3,
        'system' => 5,
        'planet' => 7,
    ]);
}

function activeNeighbourProfile(int $playerId): AiProfile
{
    return AiProfile::factory()->create([
        'player_id' => $playerId,
        'enabled' => true,
    ]);
}

/**
 * An espionage report the account holds on a neighbour one system away, whose
 * warehouse holds the given pile and whose last intel saw a few ships. The
 * report is what the planner is handed; the loot and the defender are read live
 * from the neighbour's planet, exactly as the raid decision does in play.
 *
 * @param array<string, int> $ships
 */
function activeNeighbourReport(
    int $spyPlayerId,
    int $originPlanetId,
    int $metal,
    int $crystal,
    int $deuterium,
    array $ships,
): int {
    $origin = Planet::query()->findOrFail($originPlanetId);
    $owner = User::factory()->create();

    $target = Planet::factory()->create([
        'user_id' => $owner->id,
        'galaxy' => $origin->galaxy,
        'system' => $origin->system + 1,
        'planet' => 1,
        'metal' => $metal,
        'crystal' => $crystal,
        'deuterium' => $deuterium,
        'time_last_update' => now()->getTimestamp(),
    ]);

    return EspionageReport::factory()->create([
        'user_id' => $spyPlayerId,
        'planet_user_id' => $owner->id,
        'planet_galaxy' => $target->galaxy,
        'planet_system' => $target->system,
        'planet_position' => $target->planet,
        'planet_type' => PlanetType::Planet->value,
        'ships' => $ships,
    ])->id;
}

test('a mid-phase account raids an active neighbour whose warehouse pays', function (): void {
    activeNeighbourProfile($this->currentUserId);
    activeNeighbourColony($this->currentUserId);
    $this->planetAddResources(new Resources(1_000_000, 1_000_000, 1_000_000));
    $this->planetAddUnit('small_cargo', 20);
    $this->planetAddUnit('light_fighter', 50);

    // The report saw ships on the neighbour. The ladder used to hold that class
    // back until astrophysics 23, which no mid-phase account ever reaches, so no
    // raid could fly at all; the raid is decided by what it pays instead.
    $reportId = activeNeighbourReport(
        $this->currentUserId,
        $this->currentPlanetId,
        5_000_000,
        5_000_000,
        5_000_000,
        ['small_cargo' => 4],
    );

    expect(app(RaidPlanner::class)->plan($this->currentUserId, $reportId))
        ->toBeInstanceOf(QueueableRaid::class);
});

test('a mid-phase account leaves an active neighbour that holds nothing', function (): void {
    activeNeighbourProfile($this->currentUserId);
    activeNeighbourColony($this->currentUserId);
    $this->planetAddUnit('small_cargo', 20);
    $this->planetAddUnit('light_fighter', 50);

    $reportId = activeNeighbourReport(
        $this->currentUserId,
        $this->currentPlanetId,
        0,
        0,
        0,
        ['small_cargo' => 4],
    );

    expect(app(RaidPlanner::class)->plan($this->currentUserId, $reportId))->toBeNull();
});
