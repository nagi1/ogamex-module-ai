<?php

use Modules\AI\Domain\Decision\RaidPlanner;
use Modules\AI\Models\AiProfile;
use Modules\AI\Tests\Support\AiQueueModuleTestCase;
use OGame\Models\Enums\PlanetType;
use OGame\Models\EspionageReport;
use OGame\Models\Planet;
use OGame\Models\Resources;
use OGame\Models\User;

uses(AiQueueModuleTestCase::class);

test('a mid-phase account raids an active neighbour whose warehouse pays', function (): void {
    activeNeighbourProfile($this->currentUserId);
    activeNeighbourMidPhase($this->currentUserId);
    $this->planetAddResources(new Resources(1_000_000, 1_000_000, 1_000_000));
    $this->planetAddUnit('small_cargo', 5);

    // Close to home and sitting on a full warehouse: the haul clears the fuel
    // tier and there is nothing there to lose the fleet to.
    $neighbour = activeNeighbourPlanet(1, 2, 8, 2_000_000);

    // The spy saw ships, so the target is not the probed-empty farm the old
    // ladder waited for.
    $reportId = activeNeighbourReport($this->currentUserId, $neighbour, ['light_fighter' => 2]);

    expect(app(RaidPlanner::class)->plan($this->currentUserId, $reportId))->not->toBeNull();
});

test('a mid-phase account leaves an active neighbour that holds nothing', function (): void {
    activeNeighbourProfile($this->currentUserId);
    activeNeighbourMidPhase($this->currentUserId);
    $this->planetAddResources(new Resources(1_000_000, 1_000_000, 1_000_000));
    $this->planetAddUnit('small_cargo', 5);

    // Same class of target, opposite situation: an empty warehouse a long
    // flight away, so nothing the raid brings home clears its fuel tier and
    // the account is left to fleet-save instead.
    $neighbour = activeNeighbourPlanet(5, 400, 15, 0);

    $reportId = activeNeighbourReport($this->currentUserId, $neighbour, ['light_fighter' => 2]);

    expect(app(RaidPlanner::class)->plan($this->currentUserId, $reportId))->toBeNull();
});

/** The account the planner is asked to plan for, enabled so a raid can be queued. */
function activeNeighbourProfile(int $playerId): AiProfile
{
    $profile = AiProfile::query()->where('player_id', $playerId)->first();
    if ($profile !== null) {
        $profile->forceFill(['enabled' => true])->save();

        return $profile;
    }

    return AiProfile::query()->create([
        'player_id' => $playerId,
        'enabled' => true,
        'random_seed' => 20260930,
    ]);
}

/** A second planet moves the account out of the opening, where active neighbours are lawful. */
function activeNeighbourMidPhase(int $playerId): Planet
{
    return Planet::factory()->create([
        'user_id' => $playerId,
        'galaxy' => 3,
        'system' => 400,
        'planet' => 4,
        'metal' => 0,
        'crystal' => 0,
        'deuterium' => 0,
        'time_last_update' => now()->subHour()->getTimestamp(),
    ]);
}

/** A neighbour owned by a real player — an active target, never an inactive farm. */
function activeNeighbourPlanet(int $galaxy, int $system, int $position, int $metal): Planet
{
    return Planet::factory()->create([
        'user_id' => User::factory()->create()->id,
        'galaxy' => $galaxy,
        'system' => $system,
        'planet' => $position,
        'metal' => $metal,
        'crystal' => 0,
        'deuterium' => 0,
        'time_last_update' => now()->subHour()->getTimestamp(),
    ]);
}

/**
 * The espionage report the account holds on that neighbour: host columns the
 * planner reads, with $ships standing in for what the spy saw there.
 */
function activeNeighbourReport(int $playerId, Planet $neighbour, array $ships): int
{
    return EspionageReport::factory()->create([
        'user_id' => $playerId,
        'planet_galaxy' => (int) $neighbour->galaxy,
        'planet_system' => (int) $neighbour->system,
        'planet_position' => (int) $neighbour->planet,
        'planet_type' => PlanetType::Planet->value,
        'planet_user_id' => (int) $neighbour->user_id,
        'ships' => $ships,
    ])->id;
}
